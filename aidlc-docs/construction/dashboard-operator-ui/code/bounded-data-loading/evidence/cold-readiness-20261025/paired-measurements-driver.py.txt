def fingerprint(path):
 h=hashlib.sha256()
 with path.open('rb') as f:
  for block in iter(lambda:f.read(65536),b''):h.update(block)
 return h.hexdigest()

def request(service,query):
 start=time.monotonic();r=service.request(query)
 while r.status=='pending' and time.monotonic()-start<120:r=service.request(query,wait=.05)
 assert r.complete,(query.kind,r.status,r.reason)
 return r

def measure(service,query):
 start=time.perf_counter();cpu=time.process_time();r=request(service,query)
 return r,time.perf_counter()-start,time.process_time()-cpu
