import argparse,json,subprocess,urllib.request,urllib.parse
from datetime import datetime,timedelta,timezone
from pathlib import Path
parser=argparse.ArgumentParser();parser.add_argument('--output',required=True);args=parser.parse_args()
out=Path('aidlc-docs/construction/dashboard-operator-ui/code/bounded-data-loading/evidence/page-poll-20261025')
summary=json.loads((out/'v62-20s-concurrency-summary.json').read_text())
start=datetime.fromisoformat(summary['started_at'])-timedelta(seconds=30)
credential_data=json.loads(subprocess.run(['flyctl','auth','token','--json'],capture_output=True,text=True,check=True).stdout)
credential=credential_data.get('token') or credential_data.get('access_token')
if not isinstance(credential,str):raise RuntimeError('Credential response unavailable')
authorization=credential if credential.startswith(('FlyV1 ','Bearer ')) else ('FlyV1 '+credential if credential.startswith('fm2_') else 'Bearer '+credential)
query='{__name__=~"fly_instance_cpu_(baseline|balance|throttle)",app="crypto-master",instance="6835752b711958"}'
params=urllib.parse.urlencode({'query':query,'start':start.isoformat(),'end':datetime.now(timezone.utc).isoformat(),'step':'15'})
request=urllib.request.Request('https://api.fly.io/prometheus/personal/api/v1/query_range?'+params,headers={'Authorization':authorization})
with urllib.request.urlopen(request,timeout=30) as response:data=json.load(response)
assert data['status']=='success'
(out/args.output).write_text(json.dumps(data,indent=2)+'\n')
for row in data['data']['result']:print(json.dumps({'metric':row['metric']['__name__'],'last':row.get('values',[])[-3:]}))
