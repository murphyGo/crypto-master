import argparse, base64, hashlib, json, shlex, subprocess
from pathlib import Path
p=argparse.ArgumentParser()
p.add_argument('--source', required=True)
p.add_argument('--phase', required=True)
p.add_argument('--schemas', action='store_true')
a=p.parse_args()
root=Path.cwd()
files=subprocess.check_output(['git','ls-tree','-r','--name-only',a.source],text=True).splitlines()
paths=[f for f in files if f.startswith(('src/','scripts/','config/','strategies/','trading_profiles/','docs/research/strategies/')) or f in ('start.sh','requirements.txt','pyproject.toml')]
expected={name: hashlib.sha256(subprocess.check_output(['git','show',f'{a.source}:{name}'])).hexdigest() for name in paths}
code='''
import base64, hashlib, json, os, pathlib, platform
from datetime import datetime,timezone
paths=PATHS
out={'captured_at':datetime.now(timezone.utc).isoformat(),'runtime_sha256':{name:hashlib.sha256((pathlib.Path('/app')/name).read_bytes()).hexdigest() for name in paths},'python':platform.python_version(),'processes':[],'paper_cycles':[]}
for d in pathlib.Path('/proc').iterdir():
    if not d.name.isdigit(): continue
    try:
        cmd=(d/'cmdline').read_bytes().replace(b'\\0',b' ').decode(errors='replace')
        kind='engine' if cmd.startswith('python -m src.main') else ('dashboard' if 'streamlit run src/dashboard/app.py' in cmd else '')
        if kind:
            status=(d/'status').read_text().splitlines()
            out['processes'].append({'kind':kind,'pid':int(d.name),'rss_kib':next((int(s.split()[1]) for s in status if s.startswith('VmRSS:')),None)})
    except (FileNotFoundError,PermissionError): pass
out['meminfo']={s.split(':')[0]:s.split(':')[1].strip() for s in pathlib.Path('/proc/meminfo').read_text().splitlines() if s.startswith(('MemTotal:','MemAvailable:'))}
from importlib.metadata import version
out['dependency_versions']={name:version(name) for name in ('streamlit','pandas','pydantic','pydantic-core','numpy')}
out['configuration']={k:os.environ.get(k) for k in ('TRADING_MODE','LLM_PROVIDER','CODEX_CLI_MODEL','CLAUDE_CLI_MODEL')}
activity=sorted(pathlib.Path('/data/runtime').glob('activity.*.jsonl'))
if activity:
    with activity[-1].open('rb') as h:
        h.seek(max(0,activity[-1].stat().st_size-2097152)); raw=h.read()
    for line in raw.splitlines()[1:]:
        try: row=json.loads(line)
        except ValueError: continue
        if row.get('event_type') in ('cycle_started','cycle_completed','cycle_errored'):
            out['paper_cycles'].append({k:row.get(k) for k in ('timestamp','event_type','cycle_id')})
if SCHEMAS:
    from streamlit.proto import BackMsg_pb2, ForwardMsg_pb2
    from google.protobuf import descriptor_pb2
    seen={}
    def add(desc):
        if desc.name in seen: return
        for dep in desc.dependencies: add(dep)
        seen[desc.name]=desc.serialized_pb
    add(BackMsg_pb2.DESCRIPTOR); add(ForwardMsg_pb2.DESCRIPTOR)
    ds=descriptor_pb2.FileDescriptorSet()
    for value in seen.values(): ds.file.add().ParseFromString(value)
    out['protocol_descriptor_set_base64']=base64.b64encode(ds.SerializeToString()).decode()
print(json.dumps(out))
'''.replace('PATHS',repr(paths)).replace('SCHEMAS',repr(a.schemas))
remote='python -c '+shlex.quote('import base64;exec(base64.b64decode('+repr(base64.b64encode(code.encode()).decode())+'))')
r=subprocess.run(['flyctl','ssh','console','--app','crypto-master','--machine','6835752b711958','--quiet','--command',remote],capture_output=True,text=True,timeout=120)
if r.returncode: raise RuntimeError(r.stderr[-1200:]+r.stdout[-1200:])
line=next(line for line in reversed(r.stdout.splitlines()) if line.startswith('{'))
data=json.loads(line)
data['paper_cycles']=data['paper_cycles'][-6:]
actual=data['runtime_sha256']
data['expected_source']=a.source
data['runtime_files_match']=actual==expected
data['runtime_files_checked']=len(expected)
assert actual==expected, [k for k in expected if expected[k]!=actual.get(k)]
output=root/'aidlc-docs/construction/dashboard-operator-ui/code/bounded-data-loading/evidence/shared-cpu-20261025'
output.mkdir(parents=True,exist_ok=True)
if a.schemas:
    raw=data.pop('protocol_descriptor_set_base64')
    (Path('/private/tmp')/'crypto-master-ui-protocol-20261025.pb').write_bytes(base64.b64decode(raw))
(output/(a.phase+'-runtime-verification.json')).write_text(json.dumps(data,indent=2)+'\n')
print(json.dumps({k:v for k,v in data.items() if k!='runtime_sha256'}))
